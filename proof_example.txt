c Proof format example for Project 1 
c
c This file illustrates the resolution proof format used by cdcl.py.
c
c Problem: [[1,2],[-1,2],[1,-2],[-1,-2]]  (DPLL example 2, UNSAT)
c Initial clause IDs: 1..4
c   1: { 1,  2}
c   2: {-1,  2}
c   3: { 1, -2}
c   4: {-1, -2}
c
c Format of each resolution step:
c   <new_id> <pivot> <id1> <id2> : <resolvent_literals...> 0
c
c   new_id : ID assigned to the resolvent clause
c   pivot  : the resolved literal (id1 contains -pivot, id2 contains +pivot)
c   id1    : clause containing -pivot
c   id2    : clause containing +pivot
c   after ':' : resolvent literals, terminated by 0
c               (empty resolvent = empty clause = contradiction)
c
p proof 4
5 1 2 1 : 2 0
6 1 4 3 : -2 0
7 2 6 5 : 0
